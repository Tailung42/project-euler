find_factor() returns the smallest INTEGER which when multiplies a decimal gives another INTEGER
find_factor(quotient):
    if quotient is an integer
        return 1

    decimal_part = quotient % 1
    factor = 1 / decimal_part
    if factor is approximately an integer:
        return factor
    else:
        return factor * find_factor(factor)
